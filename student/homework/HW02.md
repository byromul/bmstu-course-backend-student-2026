## Реализуйте атомарную запись

Алгоритм функции `save_game`:

1. Преобразует состояние в новую структуру и проверяет её;
2. Создаёт каталог назначения;
3. Записывает YAML во временный файл рядом с целевым;
4. Закрывает файл;
5. Заменяет целевой путь через `os.replace`.

Настройте `yaml.safe_dump` так, чтобы русские буквы оставались в файле в читаемом виде. Сбой до замены не должен обрезать прежнее сохранение.

## Реализуйте безопасную загрузку

Алгоритм функции `load_game`:

1. Читает текст во временную переменную;
2. Разбирает его через `yaml.safe_load`;
3. Проверяет JSON Schema;
4. Отдельно отклоняет `bool` вместо целого числа;
5. Проверяет `current_scene_id` через `scene_exists`;
6. Создаёт и возвращает новый `GameState`.

Отсутствующий файл, повреждённый YAML, версия `2`, лишнее или пропущенное поле, строка вместо `hp`, отрицательное число и неизвестная сцена дают `SaveGameError`. Сообщение называет причину, но не печатает полное содержимое.

Если не удалось завершить реализацию самостоятельно, замените временный `client/storage.py` полным проверенным вариантом ниже. Сначала сравните порядок проверки и место `os.replace`, затем запустите тест последовательного сохранения и загрузки.

```python
"""Атомарное YAML-сохранение состояния партии."""

from __future__ import annotations

import json
import logging
import os
from pathlib import Path

import yaml
from jsonschema import Draft202012Validator, ValidationError

from client.domain import GameState

SCHEMA_PATH = Path(__file__).resolve().parents[1] / "schemas/save.schema.json"
LOGGER = logging.getLogger(__name__)


class SaveGameError(ValueError):
    """Контролируемый отказ чтения или проверки сохранения."""


def _schema() -> dict[str, object]:
    return json.loads(SCHEMA_PATH.read_text(encoding="utf-8"))


def save_game(path: Path, state: GameState) -> None:
    document = {
        "schema_version": 1,
        "current_scene_id": state.current_scene_id,
        "player": {
            "name": state.name,
            "hp": state.hp,
            "str": state.strength,
            "def": state.defense,
        },
    }
    Draft202012Validator(_schema()).validate(document)
    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_name(path.name + ".tmp")
    serialized = yaml.safe_dump(document, allow_unicode=True, sort_keys=False)
    temporary.write_text(serialized, encoding="utf-8")
    os.replace(temporary, path)


def load_game(path: Path, scene_exists) -> GameState:
    try:
        document = yaml.safe_load(path.read_text(encoding="utf-8"))
        Draft202012Validator(_schema()).validate(document)
        player = document["player"]
        if any(isinstance(player[key], bool) for key in ("hp", "str", "def")):
            raise SaveGameError("характеристика не может быть логическим значением")
        if not scene_exists(document["current_scene_id"]):
            raise SaveGameError("сцена из сохранения отсутствует")
        return GameState(
            current_scene_id=document["current_scene_id"],
            name=player["name"],
            hp=player["hp"],
            strength=player["str"],
            defense=player["def"],
        )
    except SaveGameError:
        LOGGER.warning("Сохранение отклонено: нарушен предметный контракт")
        raise
    except FileNotFoundError as error:
        LOGGER.warning("Сохранение отклонено: файл не найден")
        raise SaveGameError("файл сохранения не найден") from error
    except (OSError, UnicodeError, yaml.YAMLError, ValidationError, KeyError, TypeError) as error:
        LOGGER.warning("Сохранение отклонено: %s", type(error).__name__)
        raise SaveGameError(f"сохранение повреждено: {type(error).__name__}") from error
```

После замены выполните `python -m unittest client.tests.test_storage.StorageTests.test_round_trip -v`: ожидается `OK`. Если тест снова красный, найдите внизу трассировки тип и сообщение исключения, затем ближайшую строку своего кода и проверьте путь `schemas/save.schema.json`.

```python
"""Позитивные и негативные AAA-тесты сохранения."""

import tempfile
import unittest
from pathlib import Path

from client.domain import GameState
from client.storage import SaveGameError, load_game, save_game


class StorageTests(unittest.TestCase):
    def test_round_trip(self) -> None:
        with tempfile.TemporaryDirectory() as temporary:
            path = Path(temporary) / "game.yaml"
            expected = GameState("WIN", "Герой", 7, 4, 3)
            save_game(path, expected)
            self.assertEqual(load_game(path, lambda scene_id: scene_id == "WIN"), expected)

    def test_broken_file_does_not_replace_current_state(self) -> None:
        with tempfile.TemporaryDirectory() as temporary:
            path = Path(temporary) / "game.yaml"
            path.write_text("schema_version: 1\nplayer: []\n", encoding="utf-8")
            current = GameState.new("Герой")
            with self.assertRaises(SaveGameError):
                load_game(path, lambda _scene_id: True)
            self.assertEqual(current, GameState.new("Герой"))

    def test_missing_file_is_rejected(self) -> None:
        with tempfile.TemporaryDirectory() as temporary:
            path = Path(temporary) / "missing.yaml"
            with self.assertRaisesRegex(SaveGameError, "не найден"):
                load_game(path, lambda _scene_id: True)

    def test_broken_yaml_is_rejected(self) -> None:
        with tempfile.TemporaryDirectory() as temporary:
            path = Path(temporary) / "game.yaml"
            path.write_text("player: [\n", encoding="utf-8")
            with self.assertRaises(SaveGameError):
                load_game(path, lambda _scene_id: True)

    def test_invalid_fields_are_rejected(self) -> None:
        base = (
            "schema_version: 1\ncurrent_scene_id: WIN\nplayer:\n"
            "  name: Герой\n  hp: 10\n  str: 3\n  def: 2\n"
        )
        samples = (
            base.replace("schema_version: 1", "schema_version: 2"),
            base.replace("  name: Герой\n", ""),
            base.replace("  hp: 10\n", "  hp: '10'\n"),
            base.replace("  hp: 10\n", "  hp: -1\n"),
            base.replace("  hp: 10\n", "  hp: true\n"),
            base + "extra: value\n",
        )
        with tempfile.TemporaryDirectory() as temporary:
            path = Path(temporary) / "game.yaml"
            for sample in samples:
                with self.subTest(sample=sample):
                    path.write_text(sample, encoding="utf-8")
                    with self.assertRaises(SaveGameError):
                        load_game(path, lambda _scene_id: True)

    def test_rejected_load_keeps_current_values(self) -> None:
        with tempfile.TemporaryDirectory() as temporary:
            path = Path(temporary) / "game.yaml"
            path.write_text("schema_version: 2\n", encoding="utf-8")
            current = GameState("SCN007", "Герой", 6, 3, 2)
            with self.assertRaises(SaveGameError):
                current = load_game(path, lambda _scene_id: True)
            self.assertEqual((current.current_scene_id, current.hp), ("SCN007", 6))

    def test_unknown_scene_is_rejected_without_document_dump(self) -> None:
        with tempfile.TemporaryDirectory() as temporary:
            path = Path(temporary) / "game.yaml"
            save_game(path, GameState("MISSING", "Герой", 7, 4, 3))
            with self.assertLogs("client.storage", level="WARNING") as logged:
                with self.assertRaisesRegex(SaveGameError, "сцена") as context:
                    load_game(path, lambda _scene_id: False)
            self.assertNotIn("Герой", str(context.exception))
            self.assertNotIn("Герой", " ".join(logged.output))


if __name__ == "__main__":
```

## Удлините одну ветку сценария

Чтобы увидеть восстановление прогресса, добавьте промежуточные сцены в `client/domain.py`. Вторая ветка потребует трёх выборов: `START → HALL → BRIDGE → LOSS`. Первый выбор из `START` по-прежнему ведёт в `WIN`.

Замените `client/domain.py` полным вариантом:

```python
"""Предметное состояние и переходы первого этапа."""

from __future__ import annotations

from dataclasses import dataclass, replace

SCENES = {
    "START": {
        "id": "START",
        "kind": "choice",
        "description": "Перед вами две двери.",
        "choices": (
            {"text": "Открыть светлую дверь", "next_scene_id": "WIN"},
            {"text": "Открыть тёмную дверь", "next_scene_id": "HALL"},
        ),
    },
    "HALL": {
        "id": "HALL",
        "kind": "choice",
        "description": "За тёмной дверью — зал с двумя проходами.",
        "choices": (
            {"text": "Пройти к мосту", "next_scene_id": "BRIDGE"},
            {"text": "Выйти через освещённый проход", "next_scene_id": "WIN"},
        ),
    },
    "BRIDGE": {
        "id": "BRIDGE",
        "kind": "choice",
        "description": "Вы у старого моста над пропастью.",
        "choices": (
            {"text": "Ступить на ветхий мост", "next_scene_id": "LOSS"},
            {"text": "Пойти по тропе вдоль берега", "next_scene_id": "WIN"},
        ),
    },
    "WIN": {
        "id": "WIN",
        "kind": "ending",
        "description": "Вы нашли выход.",
        "ending": "positive",
    },
    "LOSS": {
        "id": "LOSS",
        "kind": "ending",
        "description": "Путь завершился неудачей.",
        "ending": "negative",
    },
}


@dataclass(frozen=True)
class GameState:
    current_scene_id: str
    name: str
    hp: int
    strength: int
    defense: int

    @classmethod
    def new(cls, name: str) -> GameState:
        return cls("START", name, 10, 3, 2)


def get_scene(scene_id: str) -> dict[str, object]:
    try:
        return SCENES[scene_id]
    except KeyError as error:
        raise ValueError(f"Неизвестная сцена: {scene_id}") from error


def choose(state: GameState, number: object) -> GameState:
    if not isinstance(number, int):
        raise ValueError("Номер выбора должен быть целым числом")
    scene = get_scene(state.current_scene_id)
    choices = scene.get("choices", ())
    if not 1 <= number <= len(choices):
        raise ValueError(f"Номер выбора должен быть от 1 до {len(choices)}")
    target = choices[number - 1]["next_scene_id"]
    return replace(state, current_scene_id=target)
```

## Подключите явные команды клиента

В `client/__main__.py` добавьте команды `save`, `load` и `quit` между сценами. `save` печатает `Игра сохранена`, `load` — `Игра загружена`, а `quit` завершает программу с сообщением `Игра завершена`. Контролируемый отказ загрузки печатает `Загрузка отклонена: <причина>` и оставляет прежнюю партию.

Если при подключении команд возникла ошибка, сравните свой файл с полным рабочим вариантом. Он проверяет существование сцены по `SCENES`, поэтому принимает и новые промежуточные сцены. Результат загрузки присваивается только после успеха.

```python
"""Консольная точка входа с явным сохранением между сценами."""

import os
from pathlib import Path

from client.domain import SCENES, GameState, choose, get_scene
from client.storage import SaveGameError, load_game, save_game


def main() -> None:
    save_path = Path(os.environ.get("QUEST_SAVE_PATH", "saves/game.yaml"))
    command = input("Введите new, load или quit: ").strip().lower()
    if command == "quit":
        print("Игра завершена")
        return
    if command == "load":
        state = GameState.new("Игрок")
        try:
            state = load_game(save_path, lambda scene_id: scene_id in SCENES)
            print("Игра загружена")
        except SaveGameError as error:
            print(f"Загрузка отклонена: {error}")
    else:
        state = GameState.new(input("Имя героя: ").strip() or "Игрок")
    while True:
        scene = get_scene(state.current_scene_id)
        print(scene["description"])
        if scene["kind"] == "ending":
            print(f"Финал: {scene['ending']}")
            return
        for index, item in enumerate(scene["choices"], start=1):
            print(f"{index}. {item['text']}")
        raw = input("Номер выбора, save, load или quit: ").strip().lower()
        if raw == "quit":
            print("Игра завершена")
            return
        if raw == "save":
            save_game(save_path, state)
            print("Игра сохранена")
            continue
        if raw == "load":
            try:
                restored = load_game(save_path, lambda scene_id: scene_id in SCENES)
            except SaveGameError as error:
                print(f"Загрузка отклонена: {error}")
                continue
            state = restored
            print("Игра загружена")
            continue
        try:
            state = choose(state, int(raw))
        except (ValueError, TypeError) as error:
            print(f"Ошибка ввода: {error}")


if __name__ == "__main__":
    main()
```

Команда `quit` доступна и в начальном меню. Выход не сохраняет игру автоматически: для сохранения сначала выполните `save`.
